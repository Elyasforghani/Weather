"use client";

import { useState, useEffect, useRef } from "react";
import { Swiper, SwiperSlide } from "swiper/react";
import { FreeMode, Scrollbar } from "swiper/modules";
import "swiper/css";
import "swiper/css/free-mode";
import { MorphingText } from "@/components/ui/morphing-text";
import WeatherMap from "./weather/WeatherMap";

function getWeatherTheme(weather) {
  if (!weather) return "from-slate-950 via-slate-900 to-slate-950";
  const isNight = weather.weather?.[0]?.icon?.endsWith("n");
  const main = weather.weather?.[0]?.main?.toLowerCase() || "";

  if (isNight) return "from-indigo-950 via-slate-950 to-black";
  if (main.includes("clear") || main.includes("sun"))
    return "from-blue-700 via-sky-600 to-blue-800";
  if (main.includes("cloud"))
    return "from-slate-600 via-gray-600 to-slate-700";
  if (main.includes("rain") || main.includes("drizzle"))
    return "from-slate-700 via-indigo-800 to-slate-800";
  if (main.includes("snow"))
    return "from-slate-500 via-blue-700 to-slate-600";
  if (main.includes("thunder"))
    return "from-indigo-900 via-purple-900 to-slate-900";
  return "from-blue-800 via-sky-700 to-blue-900";
}
function SkeletonDemo() {
  return (
    <div className="flex-1 flex flex-col gap-4 animate-pulse min-h-0 w-full">
      {/* Row 1: Main Temp (Left) & Map (Right) ~ 45% Height */}
      <div className="flex flex-col md:flex-row gap-4 h-[45%] min-h-[220px]">
        {/* Main Weather Card Skeleton */}
        <div className="w-full md:w-[40%] rounded-[2rem] bg-white/5 border border-white/10 p-6 flex flex-col justify-between" />
        
        {/* Map Skeleton */}
        <div className="w-full md:w-[60%] rounded-[2rem] bg-white/5 border border-white/10" />
      </div>

      {/* Row 2: 6 Details Cards Grid ~ 18% Height */}
      <div className="grid grid-cols-3 md:grid-cols-6 gap-4 h-[18%] min-h-[90px]">
        {[...Array(6)].map((_, i) => (
          <div
            key={i}
            className="rounded-3xl bg-white/5 border border-white/10 h-full w-full"
          />
        ))}
      </div>

      {/* Row 3: Full Width Forecast Container ~ Remaining Height */}
      <div className="flex-1 min-h-[130px] rounded-[2rem] bg-white/5 border border-white/10 p-4 flex gap-4 overflow-hidden items-center">
        {[...Array(5)].map((_, i) => (
          <div
            key={i}
            className="w-40 h-full rounded-3xl bg-white/5 border border-white/5 shrink-0"
          />
        ))}
      </div>
    </div>
  );
}
export default function Home() {
  const [city, setCity] = useState("");
  const [temp, setTemp] = useState(null);
  const [loading, setLoading] = useState(false);
  const [weatherapp, setWeatherapp] = useState(null);
  const [forecast, setForecast] = useState(null);
  const [error, setError] = useState(null);

  const [isMounted, setIsMounted] = useState(false);
  const [suggestions, setSuggestions] = useState([]);
  const [showSuggestions, setShowSuggestions] = useState(false);
  const [activeIndex, setActiveIndex] = useState(-1);
  const wrapperRef = useRef(null);
  const debounceRef = useRef(null);
  const [searchedCity, setSearchedCity] = useState("");
  const [currentDate, setCurrentDate] = useState("");

  const apiKey = process.env.NEXT_PUBLIC_OPENWEATHER_API_KEY;

  useEffect(() => {
    setIsMounted(true);
  }, []);

  useEffect(() => {
    setCurrentDate(
      new Date().toLocaleDateString("en-US", {
        weekday: "long",
        month: "long",
        day: "numeric",
      })
    );
  }, []);
  useEffect(() => {
    if (!city.trim() || city.length < 2) {
      setSuggestions([]);
      return;
    }

    clearTimeout(debounceRef.current);
    debounceRef.current = setTimeout(async () => {
      try {
        const res = await fetch(
          `https://api.openweathermap.org/geo/1.0/direct?q=${encodeURIComponent(
            city
          )}&limit=5&appid=${apiKey}`
        );
        const data = await res.json();
        setSuggestions(data || []);
        setShowSuggestions(true);
        setActiveIndex(-1);
      } catch {
        setSuggestions([]);
      }
    }, 300);

    return () => clearTimeout(debounceRef.current);
  }, [city]);

  useEffect(() => {
    function handleClickOutside(e) {
      if (wrapperRef.current && !wrapperRef.current.contains(e.target)) {
        setShowSuggestions(false);
      }
    }
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  async function getTemp(selectedCity = city) {
    if (!selectedCity.trim()) return;
    setLoading(true);
    setError(null);
    setShowSuggestions(false);

    try {
      const [res1, res2] = await Promise.all([
        fetch(
          `https://api.openweathermap.org/data/2.5/weather?q=${selectedCity}&units=metric&appid=${apiKey}`
        ),
        fetch(
          `https://api.openweathermap.org/data/2.5/forecast?q=${selectedCity}&units=metric&appid=${apiKey}`
        ),
      ]);

      const weatherres = await res1.json();
      const forecastData = await res2.json();

      if (weatherres.cod !== 200) {
        throw new Error(weatherres.message || "City not found");
      }

      setTemp(weatherres.main.temp);
      setWeatherapp(weatherres);
      setForecast(forecastData);
      setSearchedCity(weatherres.name);
    } catch (err) {
      setError(err.message);
      setWeatherapp(null);
      setForecast(null);
    } finally {
      setLoading(false);
    }
  }

  function handleSuggestionClick(item) {
    setCity(item.name);
    setShowSuggestions(false);
    getTemp(item.name);
  }

  function handleKeyDown(e) {
    if (!showSuggestions || suggestions.length === 0) return;

    if (e.key === "ArrowDown") {
      e.preventDefault();
      setActiveIndex((prev) => (prev + 1) % suggestions.length);
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      setActiveIndex(
        (prev) => (prev - 1 + suggestions.length) % suggestions.length
      );
    } else if (e.key === "Enter") {
      e.preventDefault();
      if (activeIndex >= 0) {
        handleSuggestionClick(suggestions[activeIndex]);
      } else {
        getTemp();
      }
    } else if (e.key === "Escape") {
      setShowSuggestions(false);
    }
  }

  // Reusable Icon Component
  const Icon = ({ path, className = "w-5 h-5" }) => (
    <svg
      className={className}
      fill="none"
      stroke="currentColor"
      strokeWidth={1.5}
      viewBox="0 0 24 24"
    >
      <path strokeLinecap="round" strokeLinejoin="round" d={path} />
    </svg>
  );

  const texts = ["weatherApp", "search", "your city", "get forecast"];
  const bgGradient = getWeatherTheme(weatherapp);

  return (
    <main className="  overflow-hidden bg-[#0a0e1a] text-white font-sans selection:bg-cyan-500/30 relative flex flex-col">
      {/* Ambient Background Lights */}
      <div className="absolute top-0 left-1/4 w-[600px] h-[600px] bg-indigo-600/10 rounded-full blur-[120px] pointer-events-none" />
      <div className="absolute bottom-0 right-1/4 w-[500px] h-[500px] bg-cyan-500/10 rounded-full blur-[120px] pointer-events-none" />

      {/* Header & Search */}
      <header className="relative z-20 flex items-center justify-between gap-6 px-8 py-4 shrink-0 border-b border-white/5">
        <div className="hidden md:block w-[200px]">
          <h1 className="text-xl">
            <MorphingText texts={texts} />
          </h1>
        </div>

        <div className="relative flex-1 max-w-2xl mx-auto" ref={wrapperRef}>
          <div className="flex items-center gap-2 p-1.5 bg-white/5 border border-white/10 rounded-2xl backdrop-blur-xl transition-all focus-within:border-cyan-400/50 focus-within:bg-white/10">
            <div className="flex-1 relative">
              <input
                type="text"
                placeholder="Search city..."
                value={city}
                onChange={(e) => setCity(e.target.value)}
                onKeyDown={handleKeyDown}
                onFocus={() => city.length > 1 && setShowSuggestions(true)}
                className="w-full bg-transparent px-4 py-2 text-sm placeholder:text-white/30 outline-none"
              />
            </div>
            <button
              onClick={() => getTemp()}
              disabled={loading}
              className="bg-white text-slate-900 p-2.5 rounded-xl text-sm font-medium transition-all active:scale-90 disabled:opacity-50 hover:bg-cyan-300 flex items-center justify-center w-10 h-10"
            >
              {loading ? (
                <svg className="animate-spin h-5 w-5" viewBox="0 0 24 24" fill="none">
                  <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
                  <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
                </svg>
              ) : (
                <Icon path="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z" />
              )}
            </button>
          </div>

          {showSuggestions && suggestions.length > 0 && (
            <div className="absolute top-full left-0 right-0 mt-2 z-50 rounded-2xl bg-[#1e293b]/95 backdrop-blur-xl border border-white/10 shadow-2xl overflow-hidden">
              {suggestions.map((item, i) => (
                <button
                  key={`${item.lat}-${item.lon}-${i}`}
                  onClick={() => handleSuggestionClick(item)}
                  className={`w-full text-left px-4 py-3 text-sm flex items-center gap-3 transition-colors ${i === activeIndex ? "bg-white/10 text-cyan-300" : "text-white/80 hover:bg-white/5"
                    } ${i !== suggestions.length - 1 ? "border-b border-white/5" : ""}`}
                >
                  <Icon path="M17.657 16.657L13.414 20.9a1.998 1.998 0 01-2.827 0l-4.244-4.243a8 8 0 1111.314 0z M15 11a3 3 0 11-6 0 3 3 0 016 0z" className="w-4 h-4 text-white/40" />
                  <span className="flex-1 truncate">
                    {item.name}
                    {item.state && <span className="text-white/40">, {item.state}</span>}
                  </span>
                  <span className="text-xs text-white/30 bg-white/5 px-2 py-1 rounded-md">
                    {item.country}
                  </span>
                </button>
              ))}
            </div>
          )}
        </div>

        <div className="hidden md:block w-[200px] text-right text-white/40 text-xs font-medium">
          {currentDate}
        </div>
      </header>

      {/* Main Content Area (Fills remaining height) */}
      <section className="relative z-10 flex-1 flex flex-col gap-4 p-6 min-h-0">

        {error && (
          <div className="mb-2 p-4 rounded-2xl bg-red-500/10 border border-red-500/20 text-red-300 text-sm text-center shrink-0">
            {error}
          </div>
        )}

        {loading ? (
          <SkeletonDemo />
        ) : weatherapp ? (
          <>
            {/* Row 1: Main Temp (Left) & Map (Right) - 45% Height */}
            <div className="flex flex-col md:flex-row gap-4 h-[45%] min-h-0">

              {/* Main Weather Card */}
              <div className="relative overflow-hidden w-full md:w-[40%] rounded-[10px] bg-gradient-to-br from-slate-800/50 to-slate-900/50 border border-white/10 p-6 flex flex-col justify-between shadow-2xl">
                <div className="absolute -top-20 -right-20 w-64 h-64 bg-cyan-400/10 rounded-full blur-[80px] pointer-events-none" />

                <div className="relative z-10 flex items-start justify-between">
                  <div>
                    <p className="text-white/60 text-sm font-medium tracking-wide uppercase">
                      {weatherapp.name}
                    </p>
                    <p className="text-white/40 text-xs">{weatherapp.sys.country}</p>
                  </div>
                  <p className="text-white/50 text-sm capitalize font-medium">
                    {weatherapp.weather[0].description}
                  </p>
                </div>

                <div className="relative z-10 flex items-center justify-between mt-4">
                  <div className="flex items-start">
                    <h2 className="text-7xl xl:text-8xl font-thin tracking-tighter bg-gradient-to-b from-white to-white/60 bg-clip-text text-transparent">
                      {Math.round(weatherapp.main.temp)}
                    </h2>
                    <span className="text-2xl font-light mt-2 text-white/60">°C</span>
                  </div>
                  <img
                    src={`https://openweathermap.org/img/wn/${weatherapp.weather[0].icon}@4x.png`}
                    alt={weatherapp.weather[0].description}
                    className="w-32 h-32 drop-shadow-2xl -mb-4"
                  />
                </div>

                <div className="relative z-10 flex gap-6 text-sm text-white/50 border-t border-white/10 pt-3 mt-2">
                  <span>High: {Math.round(weatherapp.main.temp_max)}°</span>
                  <span>Low: {Math.round(weatherapp.main.temp_min)}°</span>
                </div>
              </div>

              {/* Map Container */}
              <div className="relative overflow-hidden w-full md:w-[60%] rounded-[2rem] bg-slate-900/50 border border-white/10 shadow-2xl p-2">
                {isMounted && (
                  <div className="w-full h-full overflow-hidden rounded-[1.5rem] relative">
                    <WeatherMap searchCity={searchedCity} />
                  </div>
                )}
              </div>
            </div>

            {/* Row 2: Details Grid - 20% Height */}
            <div className="grid grid-cols-3 md:grid-cols-6 gap-4 h-[18%] min-h-0">
              {/* Card 1 */}
              <div className="rounded-3xl bg-white/[0.03] border border-white/10 p-4 flex flex-col justify-between backdrop-blur-md">
                <div className="w-8 h-8 rounded-full bg-cyan-500/10 flex items-center justify-center text-cyan-400">
                  <Icon path="M14.828 14.828a4 4 0 01-5.656 0M9 10h.01M15 10h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" className="w-4 h-4" />
                </div>
                <div>
                  <p className="text-white/40 text-[10px] font-medium uppercase tracking-wider">Feels Like</p>
                  <p className="text-lg font-semibold mt-1">{Math.round(weatherapp.main.feels_like)}°</p>
                </div>
              </div>
              {/* Card 2 */}
              <div className="rounded-3xl bg-white/[0.03] border border-white/10 p-4 flex flex-col justify-between backdrop-blur-md">
                <div className="w-8 h-8 rounded-full bg-teal-500/10 flex items-center justify-center text-teal-400">
                  <Icon path="M9.59 4.59A2 2 0 1 1 11 8H2m10.59 11.41A2 2 0 1 0 14 16H2m15.73-8.27A2.5 2.5 0 1 1 19.5 12H2" className="w-4 h-4" />
                </div>
                <div>
                  <p className="text-white/40 text-[10px] font-medium uppercase tracking-wider">Wind</p>
                  <p className="text-lg font-semibold mt-1">{weatherapp.wind.speed} <span className="text-xs font-normal text-white/40">m/s</span></p>
                </div>
              </div>
              {/* Card 3 */}
              <div className="rounded-3xl bg-white/[0.03] border border-white/10 p-4 flex flex-col justify-between backdrop-blur-md">
                <div className="w-8 h-8 rounded-full bg-blue-500/10 flex items-center justify-center text-blue-400">
                  <Icon path="M15 12a3 3 0 11-6 0 3 3 0 016 0z M2.458 12C3.732 7.943 7.523 5 12 5c4.478 0 8.268 2.943 9.542 7-1.274 4.057-5.064 7-9.542 7-4.477 0-8.268-2.943-9.542-7z" className="w-4 h-4" />
                </div>
                <div>
                  <p className="text-white/40 text-[10px] font-medium uppercase tracking-wider">Visibility</p>
                  <p className="text-lg font-semibold mt-1">{(weatherapp.visibility / 1000).toFixed(1)} <span className="text-xs font-normal text-white/40">km</span></p>
                </div>
              </div>
              {/* Card 4 */}
              <div className="rounded-3xl bg-white/[0.03] border border-white/10 p-4 flex flex-col justify-between backdrop-blur-md">
                <div className="w-8 h-8 rounded-full bg-purple-500/10 flex items-center justify-center text-purple-400">
                  <Icon path="M3 12a9 9 0 0118 0m-9 9v-9m0 9a9 9 0 01-9-9m18 0a9 9 0 01-9 9" className="w-4 h-4" />
                </div>
                <div>
                  <p className="text-white/40 text-[10px] font-medium uppercase tracking-wider">Pressure</p>
                  <p className="text-lg font-semibold mt-1">{weatherapp.main.pressure} <span className="text-xs font-normal text-white/40">hPa</span></p>
                </div>
              </div>
              {/* Card 5 */}
              <div className="rounded-3xl bg-gradient-to-br from-amber-500/10 to-transparent border border-white/10 p-4 flex flex-col justify-between backdrop-blur-md">
                <div className="w-8 h-8 rounded-full bg-amber-500/20 flex items-center justify-center text-amber-400">
                  <Icon path="M12 3v1m0 16v1m9-9h-1M4 12H3m15.364 6.364l-.707-.707M6.343 6.343l-.707-.707m12.728 0l-.707.707M6.343 17.657l-.707.707M16 12a4 4 0 11-8 0 4 4 0 018 0z" className="w-4 h-4" />
                </div>
                <div>
                  <p className="text-white/40 text-[10px] font-medium uppercase tracking-wider">Sunrise</p>
                  <p className="text-sm font-semibold mt-1">
                    {new Date(weatherapp.sys.sunrise * 1000).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}
                  </p>
                </div>
              </div>
              {/* Card 6 */}
              <div className="rounded-3xl bg-gradient-to-br from-orange-500/10 to-transparent border border-white/10 p-4 flex flex-col justify-between backdrop-blur-md">
                <div className="w-8 h-8 rounded-full bg-orange-500/20 flex items-center justify-center text-orange-400">
                  <Icon path="M17 18a5 5 0 00-10 0M12 2v7m4-4l-1.5 1.5M8 5l1.5 1.5m-7 6h1m18 0h1M5.6 5.6l.7.7m11.4-.7l-.7.7" className="w-4 h-4" />
                </div>
                <div>
                  <p className="text-white/40 text-[10px] font-medium uppercase tracking-wider">Sunset</p>
                  <p className="text-sm font-semibold mt-1">
                    {new Date(weatherapp.sys.sunset * 1000).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}
                  </p>
                </div>
              </div>
            </div>

            {/* Row 3: Full Width Forecast Swiper - 35% Height */}
            <div className="flex-1 min-h-0 flex flex-col gap-2 bg-white/[0.02] border border-white/10 rounded-[2rem] p-4 backdrop-blur-md">
              <div className="flex items-center justify-between px-2 shrink-0">
                <p className="text-white/70 text-sm font-semibold uppercase tracking-wider">
                  5-Day Forecast
                </p>
                <Icon path="M9 5l7 7-7 7" className="w-4 h-4 text-white/30" />
              </div>

              <div className="flex-1 min-h-0 forecast-list">
                {(() => {
                  const dailyData = {};
                  forecast?.list?.forEach((item) => {
                    const date = item.dt_txt.split(" ")[0];
                    if (!dailyData[date]) {
                      dailyData[date] = { temps: [], weather: item.weather[0], pop: item.pop, dt: item.dt };
                    }
                    dailyData[date].temps.push(item.main.temp);
                    dailyData[date].pop = Math.max(dailyData[date].pop, item.pop);
                  });

                  const days = Object.entries(dailyData).map(([date, data]) => ({
                    date,
                    min: Math.min(...data.temps),
                    max: Math.max(...data.temps),
                    weather: data.weather,
                    pop: data.pop,
                    dt: data.dt,
                  }));

                  return (
                    <Swiper
                      modules={[FreeMode, Scrollbar]}
                      scrollbar={{ draggable: true, hide: true, dragSize: 40 }}
                      freeMode={{ enabled: true, momentum: true, momentumRatio: 0.6, momentumVelocityRatio: 0.4, minimumVelocity: 0.02 }}
                      grabCursor={true}
                      speed={450}
                      touchRatio={1.8}
                      threshold={5}
                      resistance={true}
                      resistanceRatio={0.6}
                      watchSlidesProgress={true}
                      preventClicks={false}
                      preventClicksPropagation={false}
                      slidesPerView={"auto"}
                      spaceBetween={16}
                      style={{ height: '100%', paddingRight: "20px" }}
                    >
                      {days.map((day, i) => (
                        <SwiperSlide key={i} style={{ width: "160px", height: '100%' }}>
                          <div className="h-full w-full rounded-3xl bg-white/[0.03] border border-white/10 p-4 flex flex-col items-center justify-center gap-2 select-none hover:bg-white/[0.06] transition-all duration-200">
                            <p className="text-xs text-white/50 font-bold uppercase tracking-wider">
                              {new Date(day.date).toLocaleDateString("en-US", { weekday: "short" })}
                            </p>
                            <img
                              src={`https://openweathermap.org/img/wn/${day.weather.icon}@2x.png`}
                              alt={day.weather.description}
                              className="w-12 h-12 drop-shadow-lg my-1"
                            />
                            <div className="flex flex-col items-center leading-none gap-1">
                              <span className="text-lg font-bold text-white">{Math.round(day.max)}°</span>
                              <span className="text-xs text-white/40 font-medium">{Math.round(day.min)}°</span>
                            </div>
                            {day.pop > 0.05 && (
                              <div className="flex items-center gap-1 text-[10px] text-cyan-300 bg-cyan-400/10 px-2 py-1 rounded-full font-semibold mt-1">
                                <Icon path="M12 2.69l5.66 5.66a8 8 0 1 1-11.31 0z" className="w-3 h-3" />
                                {(day.pop * 100).toFixed(0)}%
                              </div>
                            )}
                          </div>
                        </SwiperSlide>
                      ))}
                    </Swiper>
                  );
                })()}
              </div>
            </div>
          </>
        ) : (
          <div className="flex-1 flex flex-col items-center justify-center text-white/30 gap-4">
            <div className="w-20 h-20 rounded-full bg-white/5 border border-white/10 flex items-center justify-center">
              <svg className="w-10 h-10 text-white/40" fill="none" stroke="currentColor" strokeWidth={1} viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" d="M3 15a4 4 0 004 4h9a5 5 0 10-.1-9.999 5.002 5.002 0 10-9.78 2.096A4.001 4.001 0 003 15z" />
              </svg>
            </div>
            <p className="text-sm font-medium">Enter a city to see the weather</p>
          </div>
        )}
      </section>
    </main>
  );
}
