'use client'
import React, { useState } from 'react'
import useFetch from './useFetch'

const CustomSpinner = () => (
  <div className="flex justify-center items-center min-h-screen bg-slate-900 text-white">
    <div className="animate-spin rounded-full h-12 w-12 border-t-4 border-b-4 border-blue-500"></div>
  </div>
);

export default function Page() {
  const [city, setCity] = useState('San Francisco')
  const [inputValue, setInputValue] = useState('')

  const { data, loading } = useFetch(`http://api.weatherapi.com/v1/forecast.json?key=5da60fdb8e1e409191a224035261208&q=${city}&aqi=no&days=7`)
  
  if (!data && loading) return <CustomSpinner />;

  const handleSearch = () => {
    if(inputValue.trim()) setCity(inputValue.trim())
  }
  
  const handleKeyDown = (e) => {
    if (e.key === 'Enter') handleSearch()
  }

  // Helper to format time from "2023-10-25 14:00" to "14:00"
  const formatTime = (timeString) => {
    const time = timeString.split(' ')[1];
    const hour = parseInt(time.split(':')[0]);
    if (hour === 0) return '12 AM';
    if (hour < 12) return `${hour} AM`;
    if (hour === 12) return '12 PM';
    return `${hour - 12} PM`;
  };

  // Helper to format date from "2023-10-25" to "Wed"
  const formatDay = (dateString) => {
    const date = new Date(dateString);
    return date.toLocaleDateString('en-US', { weekday: 'short' });
  };

  return (
    <div className="min-h-screen bg-slate-900 text-white font-sans flex flex-col items-center pb-20">
      
      {/* Top Search Bar */}
      <div className="w-full max-w-5xl p-6 flex justify-between items-center">
        <h1 className="text-xl font-bold text-slate-400">Weather App</h1>
        <div className="flex gap-2">
          <input 
            type="text" 
            className="bg-slate-800 text-white px-4 py-2 rounded-full text-sm focus:outline-none focus:ring-2 focus:ring-blue-500 w-64" 
            onKeyDown={handleKeyDown} 
            value={inputValue} 
            onChange={(e) => setInputValue(e.target.value)} 
            placeholder="Search city..."
          />
          <button onClick={handleSearch} className="bg-blue-600 hover:bg-blue-700 px-4 py-2 rounded-full text-sm font-semibold">Search</button>
        </div>
      </div>

      {loading && <p className="text-blue-400 animate-pulse mb-4">Updating forecast...</p>}

      {data?.error ? (
        <div className="mt-20 text-center">
          <p className="text-2xl text-red-400 mb-2">🚫 {data.error.message}</p>
          <p className="text-slate-400">Please try another city.</p>
        </div>
      ) : data ? (
        <div className="w-full max-w-5xl px-6 grid grid-cols-1 lg:grid-cols-3 gap-6">
          
          {/* LEFT COLUMN: Current & Hourly (Takes 2 cols on desktop) */}
          <div className="lg:col-span-2 space-y-6">
            
            {/* Current Weather Header */}
            <div className="bg-slate-800 rounded-2xl p-6 flex items-center justify-between">
              <div>
                <h2 className="text-2xl font-semibold">{data.location.name}, {data.location.region}</h2>
                <p className="text-slate-400 text-sm">{new Date().toLocaleDateString('en-US', { weekday: 'long', month: 'long', day: 'numeric' })}</p>
                <div className="flex items-end mt-2">
                  <span className="text-6xl font-light">{Math.round(data.current.temp_c)}°</span>
                  <span className="text-slate-400 ml-2 mb-2">{data.current.condition.text}</span>
                </div>
                <div className="flex gap-4 text-sm text-slate-400 mt-2">
                  <span>H: {Math.round(data.forecast.forecastday[0].day.maxtemp_c)}°</span>
                  <span>L: {Math.round(data.forecast.forecastday[0].day.mintemp_c)}°</span>
                  <span>💧 {data.current.humidity}%</span>
                  <span>🌬️ {data.current.wind_kph} kph</span>
                </div>
              </div>
              <img src={`https:${data.current.condition.icon}`} alt="weather icon" className="w-24 h-24 drop-shadow-lg" />
            </div>

            {/* Hourly Forecast */}
            <div className="bg-slate-800 rounded-2xl p-6">
              <h3 className="text-slate-400 font-semibold mb-4 uppercase text-xs tracking-wide">Hourly Forecast</h3>
              <div className="flex gap-6 overflow-x-auto pb-2">
                {/* Get the next 12 hours starting from the current hour */}
                {data.forecast.forecastday[0].hour.slice(new Date().getHours(), new Date().getHours() + 12).map((hour, index) => (
                  <div key={index} className="flex flex-col items-center gap-2 flex-shrink-0">
                    <span className="text-xs text-slate-400">{index === 0 ? 'Now' : formatTime(hour.time)}</span>
                    <img src={`https:${hour.condition.icon}`} alt="hour icon" className="w-8 h-8" />
                    <span className="font-medium">{Math.round(hour.temp_c)}°</span>
                    <span className="text-xs text-blue-400">{hour.chance_of_rain}%</span>
                  </div>
                ))}
              </div>
            </div>

            {/* Daily Forecast List */}
            <div className="bg-slate-800 rounded-2xl p-6">
              <h3 className="text-slate-400 font-semibold mb-4 uppercase text-xs tracking-wide">7-Day Forecast</h3>
              <div className="flex flex-col gap-3">
                {data.forecast.forecastday.map((day, index) => (
                  <div key={day.date} className="flex items-center justify-between border-b border-slate-700 pb-3 last:border-0 last:pb-0">
                    <span className="w-12 font-medium">{index === 0 ? 'Today' : formatDay(day.date)}</span>
                    <img src={`https:${day.day.condition.icon}`} alt="day icon" className="w-8 h-8" />
                    <span className="text-blue-400 text-sm w-12">{day.day.daily_chance_of_rain}%</span>
                    <div className="flex items-center gap-2 w-32">
                      <span className="text-slate-500 text-sm">{Math.round(day.day.mintemp_c)}°</span>
                      <div className="w-full bg-slate-700 rounded-full h-1.5">
                        <div className="bg-gradient-to-r from-blue-500 to-amber-500 h-1.5 rounded-full" style={{ width: '100%' }}></div>
                      </div>
                      <span className="text-white text-sm">{Math.round(day.day.maxtemp_c)}°</span>
                    </div>
                  </div>
                ))}
              </div>
            </div>
          </div>

          {/* RIGHT COLUMN: Radar Placeholder & Extra Details */}
          <div className="space-y-6">
            <div className="bg-slate-800 rounded-2xl overflow-hidden h-80 relative">
              {/* Fake Radar Visual */}
              <div className="absolute inset-0 bg-gradient-to-br from-slate-900 to-blue-900 flex items-center justify-center">
                <div className="w-40 h-40 bg-green-500/20 rounded-full blur-2xl animate-pulse"></div>
                <div className="absolute w-24 h-24 bg-blue-500/40 rounded-full blur-xl"></div>
              </div>
              <div className="absolute bottom-0 left-0 right-0 p-4 bg-gradient-to-t from-slate-900 to-transparent">
                <p className="text-sm font-semibold mb-1">Precipitation Radar</p>
                <button className="text-xs text-blue-400 hover:text-blue-300">View Full Radar &rarr;</button>
              </div>
            </div>

            {/* Extra Details Grid */}
            <div className="bg-slate-800 rounded-2xl p-6 grid grid-cols-2 gap-4">
              <div>
                <p className="text-slate-400 text-xs uppercase">Feels Like</p>
                <p className="text-2xl font-light">{Math.round(data.current.feelslike_c)}°</p>
              </div>
              <div>
                <p className="text-slate-400 text-xs uppercase">UV Index</p>
                <p className="text-2xl font-light">{data.current.uv}</p>
              </div>
              <div>
                <p className="text-slate-400 text-xs uppercase">Visibility</p>
                <p className="text-2xl font-light">{data.current.vis_km} km</p>
              </div>
              <div>
                <p className="text-slate-400 text-xs uppercase">Pressure</p>
                <p className="text-2xl font-light">{data.current.pressure_mb} mb</p>
              </div>
            </div>
          </div>

        </div>
      ) : null}

      {/* Bottom Navigation Bar */}
      <div className="fixed bottom-0 left-0 right-0 bg-slate-800/80 backdrop-blur-md border-t border-slate-700">
        <div className="max-w-5xl mx-auto flex justify-around items-center h-16">
          <button className="flex flex-col items-center text-blue-400">
            <span className="text-xl">🏠</span>
            <span className="text-xs mt-1">Home</span>
          </button>
          <button className="flex flex-col items-center text-slate-400 hover:text-white">
            <span className="text-xl">🗺️</span>
            <span className="text-xs mt-1">Map</span>
          </button>
          <button className="flex flex-col items-center text-slate-400 hover:text-white">
            <span className="text-xl">📊</span>
            <span className="text-xs mt-1">Stats</span>
          </button>
          <button className="flex flex-col items-center text-slate-400 hover:text-white">
            <span className="text-xl">⚙️</span>
            <span className="text-xs mt-1">Settings</span>
          </button>
        </div>
      </div>

    </div>
  )
}
