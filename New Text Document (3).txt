'use client'
import { useState } from 'react'
import useFetch from './useFetch'
import SearchBar from './components/SearchBar'
import CurrentWeather from './components/CurrentWeather'
import HourlyForecast from './components/HourlyForecast'
import DailyForecast from './components/DailyForecast'
import RadarCard from './components/RadarCard'
import BottomNav from './components/BottomNav'

export default function Page() {
  const [city, setCity] = useState('San Francisco')
  const [inputValue, setInputValue] = useState('')

  const { data, loading } = useFetch(`http://api.weatherapi.com/v1/forecast.json?key=5da60fdb8e1e409191a224035261208&q=${city}&aqi=no&days=7`)


  if (!data && loading) return <CustomSpinner />

  const handleSearch = () => {
    if (inputValue.trim()) setCity(inputValue.trim())
  }

  // Get next 12 hours from current time
  const currentHour = new Date().getHours()
  const hourlyData = data?.forecast?.forecastday[0]?.hour.slice(currentHour, currentHour + 12) || []

  return (
    <div className="min-h-screen bg-gradient-to-b from-slate-900 via-slate-900 to-slate-950 pb-24">
      <div className="max-w-5xl mx-auto pt-6 space-y-6">

        {/* Top Search */}
        <SearchBar
          value={inputValue}
          onChange={setInputValue}
          onSearch={handleSearch}
        />

        {loading && (
          <p className="text-center text-blue-400 animate-pulse text-sm">Updating forecast...</p>
        )}

        {data?.error ? (
          <div className="mt-20 text-center">
            <p className="text-2xl text-red-400 mb-2">🚫 {data.error.message}</p>
            <p className="text-slate-400">Please try another city.</p>
          </div>
        ) : data ? (
          <>
            {/* Current Weather */}
            <CurrentWeather data={data} />

            {/* Hourly Forecast */}
            {/* <HourlyForecast hours={hourlyData} /> */}

            {/* Bottom Grid: Daily + Radar */}
            <div className="px-4 grid grid-cols-1 lg:grid-cols-5 gap-6 max-w-4xl mx-auto">
              <div className="lg:col-span-3">
                {/* <DailyForecast days={data.forecast.forecastday} /> */}
              </div>
              <div className="lg:col-span-2">
                {/* <RadarCard
                  lat={data.location.lat}
                  lon={data.location.lon}
                  cityName={data.location.name} 
                /> */}
              </div>
            </div>
          </>
        ) : null}
      </div>

      <BottomNav />
    </div>
  )
}

const CustomSpinner = () => (
  <div className="flex justify-center items-center min-h-screen bg-slate-900 text-white">
    <div className="animate-spin rounded-full h-12 w-12 border-t-4 border-b-4 border-blue-500" />
  </div>
)
